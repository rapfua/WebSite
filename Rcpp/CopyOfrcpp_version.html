// Structure to represent an edge in the graph
struct edge {
  int u, v; 
  double w; 
  edge() {}
  edge(int _u, int _v, double _w) : u(_u), v(_v), w(_w) {}
};

// Structure to represent a node containing edges
struct node {
  std::vector<edge> edges;
};

// Helper for hashing pairs
struct pair_hash {
  inline std::size_t operator()(const std::pair<int,int>& v) const {
    return v.first * 31 + v.second;
  }
};

// Graph structure
class graph {
public:
  int n, m;
  std::vector<node> nodes;
  
  //std::unordered_map<int, int, double, pair_hash> edgesMB;
  std::unordered_map<std::pair<int, int>, double, pair_hash> edgesMB;

  
  // Constructor: Create a graph from a DataFrame of edges
  graph(Rcpp::DataFrame df) {

    Rcpp::Rcout << "Constructing graph" << std::endl;

    Rcpp::IntegerVector from = df["from"];
    Rcpp::IntegerVector to = df["to"];
    Rcpp::NumericVector weight = df["weight"];
    

     
    m = from.size();
    n = std::max(*std::max_element(from.begin(), from.end()), *std::max_element(to.begin(), to.end())) + 1;  // Number of nodes
    nodes.resize(n);
    
    for (int i = 0; i < m; i++) {
      int u = from[i];
      int v = to[i];
      double w = weight[i];
      add_edge(u, v, w);
    }

    getMB();
  }
  
  // Returns the adjacency matrix of the metric backbone
  arma::mat get_MB_adjacency_matrix() {
    // Initialize an n x n matrix with zeros
    arma::mat A = arma::zeros<arma::mat>(n, n);
    
    // Fill the adjacency matrix using the edgesMB map
    for (const auto& [edge_pair, weight] : edgesMB) {

      // std::cout << weight << std::endl;
      int u = edge_pair.first;
      int v = edge_pair.second;
      A(u, v) = weight;
      A(v, u) = weight;  // Since the graph is undirected
    }
    
    return A;
  }
  
  
  // Adds an edge between two nodes
  void add_edge(int u, int v, double w) {
    nodes[u].edges.emplace_back(u, v, w);
        //nodes[v].edges.emplace_back(v, u, w);

  }
  
  
  // Builds the metric backbone
  void getMB() {

    
    Rcpp::Rcout << "Constructing its MB" << std::endl;
    std::vector<int> nodesToTry(n);
    std::iota(nodesToTry.begin(), nodesToTry.end(), 0);
    
    for (int s : nodesToTry) {
      if (s % (n / 5) == 0) {
        Rcpp::Rcout << "MB " << s << " / " << n << std::endl;
      }
  
      std::vector<std::pair<int, double>> parent(n, {-1, -1});
      getSPT(s, parent);
      for (int u = 0; u < n; ++u) {
        auto [v, w] = parent[u];
        if (v != -1) {
          edgesMB[{u, v}] = w;
        }
      }
    }
    Rcpp::Rcout << "MB_constructed" << std::endl << std::endl;
  }
  
  // Builds a shortest path tree (SPT) using Dijkstra's algorithm
  void getSPT(int s, std::vector<std::pair<int, double>>& parent) {
    std::vector<double> dist(n, 1e15);
    std::vector<bool> visited(n, false);
    std::priority_queue<std::pair<double, int>, std::vector<std::pair<double, int>>, std::greater<>> pq;
    dist[s] = 0.0;
    pq.emplace(0.0, s);
    
    while (!pq.empty()) {
      int cur = pq.top().second;
      pq.pop();
      if (visited[cur]) continue;
      visited[cur] = true;
      
      for (const auto& e : nodes[cur].edges) {
        if (dist[e.v] > dist[cur] + e.w) {
          dist[e.v] = dist[cur] + e.w;
          parent[e.v] = {e.u, e.w};
          pq.emplace(dist[e.v], e.v);
        }
      }
    }
  }
  

};




// [[Rcpp::export]]
arma::mat get_metric_backbone_adjacency_matrix(Rcpp::DataFrame df) {
  // Create a graph instance from the DataFrame
  graph G(df);
  
  // Return the adjacency matrix of the metric backbone
  return G.get_MB_adjacency_matrix();
}



// Part 1: Add Edges Based on Gaussian Weight Matrix
// [[Rcpp::export]]
Rcpp::DataFrame add_edges(const arma::mat& W) {
  Rcpp::Rcout << "Adding edges to a df that will be passed to a graph constructor" << std::endl;
  int n = W.n_rows;
  std::vector<int> from;
  std::vector<int> to;
  std::vector<double> weights;
  
  for (int i = 0; i < n; ++i) {
    if (i % (n / 5) == 0) {
      Rcpp::Rcout << "Processing node " << i << " / " << n << std::endl;
    }
    
    for (int j = i + 1; j < n; ++j) {
      double w = W(i, j);
      if (w > 0) {
        from.push_back(i);
        to.push_back(j);
        weights.push_back(1.0 / w - 1.0);
      }
    }
  }

  Rcpp::Rcout << std::endl;
  
  return Rcpp::DataFrame::create(Rcpp::Named("from") = from,
                                 Rcpp::Named("to") = to,
                                 Rcpp::Named("weight") = weights);
}



void add_edges_to_file(const arma::mat& W, const std::string& filename) {
    std::ofstream ofs(filename);
    if (!ofs.is_open()) {
        std::cerr << "Error: Could not open the file for writing!" << std::endl;
        return;
    }
  
    int n = W.n_rows;  // Number of rows (nodes)
  
    ofs << "Writing edges to the file " << filename << std::endl;
  
    for (int i = 0; i < n; ++i) {
        if (i % (n / 5) == 0) {
            std::cout << "Processing node " << i << " / " << n << std::endl;
        }

        for (int j = i + 1; j < n; ++j) {
            double w = W(i, j);
            if (w > 0) {
                ofs << i << " " << j << " " << 1.0 / w - 1.0 << std::endl;
            }
        }
    }

    ofs.close();
    std::cout << "Edges written to " << filename << std::endl;
}

  Rcpp::Environment mclust("package:mclust");
  Rcpp::Function adjustedRandIndex = mclust["adjustedRandIndex"];
  
  // Call R's adjustedRandIndex function
  Rcpp::NumericVector result = adjustedRandIndex(true_labels, pred_labels);
  
  // Return the result as a double
  return Rcpp::as<double>(result);
}

def compute_accuracy(true_labels, pred_labels):
    """ Compute accuracy based on label matching using the Hungarian algorithm. """
    cm = confusion_matrix(true_labels, pred_labels)
    row_ind, col_ind = linear_sum_assignment(cm.max() - cm)
    correct = cm[row_ind, col_ind].sum()
    accuracy = correct / len(true_labels)
    return accuracy

def get_Gaussian_weight_matrix(X, n_neighbors):
    Z = gl.weightmatrix.knn(data=X, k=n_neighbors)  # Gaussian similarity measure
    A = (Z + Z.T) / 2
    return A

def remove_isolated_nodes_sparse(adj_matrix):
    # Sum rows and columns to identify isolated nodes
    row_sum = adj_matrix.sum(axis=1).A1  # .A1 converts the sparse matrix to a 1D NumPy array
    col_sum = adj_matrix.sum(axis=0).A1  # .A1 converts the sparse matrix to a 1D NumPy array
    
    # Find indices of isolated nodes (where both row and column sums are zero)
    isolated_nodes = np.where((row_sum == 0) & (col_sum == 0))[0]
    
    # Remove the isolated nodes (rows and columns)
    mask = np.ones(adj_matrix.shape[0], dtype=bool)
    mask[isolated_nodes] = False
    
    new_adj_matrix = adj_matrix[mask, :][:, mask]
    
    return new_adj_matrix, isolated_nodes

def get_ari_accuracy_multiple_simulations(X_train_flat, n_neighbors, true_labels, n_sim, random_states):
    ari_metric_backbone_list = []
    ari_original_list = []
    acc_metric_backbone_list = []
    acc_original_list = []

    for _ in range(n_sim):
        # Get ARI for metric backbone
        ari_mb, accuracy_mb = get_similarity_and_accuracy(X_train_flat, n_neighbors, true_labels, b_original=False, random_state=random_states[_])
        ari_metric_backbone_list.append(ari_mb)
        acc_metric_backbone_list.append(accuracy_mb)

        # Get ARI for original graph
        ari_orig, accuracy_orig = get_similarity_and_accuracy(X_train_flat, n_neighbors, true_labels, b_original=True, random_state=random_states[_])
        ari_original_list.append(ari_orig)
        acc_original_list.append(accuracy_orig)

    # Calculate average and standard deviation for both ARIs
    ari_mb_avg = np.mean(ari_metric_backbone_list)
    ari_mb_std = np.std(ari_metric_backbone_list)
    acc_mb_avg = np.mean(acc_metric_backbone_list)
    acc_mb_std = np.std(acc_metric_backbone_list)

    ari_orig_avg = np.mean(ari_original_list)
    ari_orig_std = np.std(ari_original_list)
    acc_orig_avg = np.mean(acc_original_list)
    acc_orig_std = np.std(acc_original_list)

    return ari_mb_avg, ari_mb_std, acc_mb_avg, acc_mb_std, ari_orig_avg, ari_orig_std, acc_orig_avg, acc_orig_std
